public class CommonService : ICommonService
{

    public HttpResponseMessage PrintReportFileToClientSide(ReportViewer reportViewer, string fileName, string printType = "PDF")
    {
        try
        {
            if (printType == "Excel" || printType == "EXCEL")
            {
                printType = "EXCELOPENXML";
            }
            byte[] bytes = reportViewer.LocalReport.Render(printType, null, out string mimeType, out string encoding, out string extension, out string[] streamids, out Warning[] warnings);

            if (bytes == null || bytes.Length == 0)
            {
                return new HttpResponseMessage(HttpStatusCode.NotFound)
                {
                    Content = new StringContent("Report generation returned empty data.")
                };
            }
           
            string fileExtension = printType.Equals("EXCELOPENXML", StringComparison.OrdinalIgnoreCase) ? "xlsx" : "pdf";
            string fullFileName = $"{fileName}.{fileExtension}";
            string contentType = printType.Equals("EXCELOPENXML", StringComparison.OrdinalIgnoreCase)
                ? "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
                : "application/pdf";

            HttpResponseMessage response = new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = new ByteArrayContent(bytes)
            };

            response.Content.Headers.ContentType = new System.Net.Http.Headers.MediaTypeHeaderValue(contentType);
            response.Content.Headers.Add("Content-Disposition", $"attachment; filename*=UTF-8''{Uri.EscapeDataString(fullFileName)}");

            response.Headers.Add("Access-Control-Expose-Headers", "Content-Disposition");
            return response;
        }
        catch (Exception ex)
        {
            return new HttpResponseMessage(HttpStatusCode.InternalServerError)
            {
                Content = new StringContent("Error generating report: " + ex.Message)
            };
        }
    }

}

Function ConvertAboveThousand(value As Integer) As String

	  Dim result As String = ""

	  Dim millions As Integer = value \ 1000000
	  Dim thousands As Integer = (value Mod 1000000) \ 1000
	  Dim remainder As Integer = value Mod 1000

	  If millions &gt; 0 Then
	  result &amp;= ConvertBelowThousand(millions) &amp; " مليون"

	  If thousands &gt; 0 Or remainder &gt; 0 Then
	  result &amp;= " و "
	  End If
	  End If

	  If thousands &gt; 0 Then
	  result &amp;= ConvertBelowThousand(thousands) &amp; " ألف"

	  If remainder &gt; 0 Then
	  result &amp;= " و "
	  End If
	  End If

	  If remainder &gt; 0 Then
	  result &amp;= ConvertBelowThousand(remainder)
	  End If

	  Return result.Trim()

End Function

		Function ConvertBelowThousand(value As Integer) As String
		Dim ones() As String = {"", "واحد", "اثنان", "ثلاثة", "أربعة", "خمسة", "ستة", "سبعة", "ثمانية", "تسعة"}
		Dim tens() As String = {"", "عشرة", "عشرون", "ثلاثون", "أربعون", "خمسون", "ستون", "سبعون", "ثمانون", "تسعون"}
		Dim teens() As String = {"عشرة", "أحد عشر", "اثنا عشر", "ثلاثة عشر", "أربعة عشر", "خمسة عشر", "ستة عشر", "سبعة عشر", "ثمانية عشر", "تسعة عشر"}
		Dim hundreds() As String = {"", "مائة", "مائتان", "ثلاثمائة", "أربعمائة", "خمسمائة", "ستمائة", "سبعمائة", "ثمانمائة", "تسعمائة"}
		Dim result As String = ""

		Dim hundredPart As Integer = value \ 100
		Dim tenPart As Integer = (value Mod 100) \ 10
		Dim onePart As Integer = value Mod 10

		' Handle hundreds
		If hundredPart &gt; 0 Then
		result &amp;= hundreds(hundredPart)
		If tenPart &gt; 0 Or onePart &gt; 0 Then result &amp;= " و "
		End If

		' Handle tens and ones
		If tenPart = 1 Then
		result &amp;= teens(onePart) ' Teens include both tens and ones
		Else
		' Handle ones first
		If onePart &gt; 0 Then
		result &amp;= ones(onePart)
		If tenPart &gt; 0 Then result &amp;= " و " ' Add "و" between ones and tens
		End If
		' Handle tens
		If tenPart &gt; 0 Then result &amp;= tens(tenPart)
		End If

		Return result.Trim()
		End Function
