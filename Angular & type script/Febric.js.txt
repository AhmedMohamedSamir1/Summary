-- the behavior of fabric: there are two canvas
	- Lower Canvas (your original), the one written in html page
	- Upper Canvas (auto-created), Created automatically by Fabric
		Used for: Selection borders, Dragging previews, Mouse interactions, Controls (resize handles, rotation, etc.)

	- <div class="canvas-container">
		<canvas class="lower-canvas"></canvas>
   		<canvas class="upper-canvas"></canvas>
	  </div>


-- initialize canvas
	private canvas!: fabric.Canvas;
	canvas = new fabric.Canvas('myCanvas');

-- fabric.Canvas members
	canvas.getElement();    // return HTMLCanvasElement
	canvas.setDimensions({ width, height });
	canvas.renderAll();


-- to create object and add it to canvas
	const rect = new fabric.Rect({
  	   left: 100,
  	   top: 100,
  	   width: 200,
  	   height: 100,
  	   fill: 'red'
	});
	this.canvas.add(rect);

-- Rendering Engine
	- it rerenders everthing when any change occurs like moving any element


-- properties
	- left
	- top
	- scalex
	- scaley
	- angle
	- opacity
	- selectable: true // this element is selectable, if false the user can't move it

-- for Z-Index / Layering
	- canvas.bringToFront(obj);
	- canvas.sendToBack(obj);


-- to Group objects together
	- const group = new fabric.Group([rect, text]);
	- canvas.add(group);
	// why using it: move group together as on element

-- Events
	canvas.on('object:selected', (e) => {
   	   console.log(e.target);
	});




-- to save the design [ serialization ]
	- const json = canvas.toJSON();

-- to load the design
	- canvas.loadFromJSON(json);


-- to convert the design into image [ exporting ]
	- const image = canvas.toDataURL();
